set(_tool_defs)
if(VMAF_HAVE_STRSEP)
    list(APPEND _tool_defs HAVE_STRSEP)
endif()

set(_tool_sources vmaf.c cli_parse.c y4m_input.c vidinput.c yuv_input.c)
set(VMAF_GETOPT_SOURCES)
set(VMAF_GETOPT_INCLUDES)
if(NOT VMAF_HAVE_GETOPT_LONG)
    set(VMAF_GETOPT_SOURCES "${VMAF_ROOT}/src/compat/getopt/getopt.c")
    set(VMAF_GETOPT_INCLUDES "${VMAF_ROOT}/src/compat/getopt")
endif()
set(VMAF_GETOPT_SOURCES ${VMAF_GETOPT_SOURCES} PARENT_SCOPE)
set(VMAF_GETOPT_INCLUDES ${VMAF_GETOPT_INCLUDES} PARENT_SCOPE)

add_executable(vmaf_cli ${_tool_sources} ${VMAF_GETOPT_SOURCES})
set_target_properties(vmaf_cli PROPERTIES OUTPUT_NAME vmaf PDB_NAME vmaf_cli)
target_compile_definitions(vmaf_cli PRIVATE HAVE_GETOPT_H ${_tool_defs})
target_include_directories(vmaf_cli PRIVATE ${VMAF_GETOPT_INCLUDES})
target_link_libraries(vmaf_cli PRIVATE ${VMAF_LINK_LIB} vmaf_threads vmaf_vulkan_deps)
if(VMAF_ENABLE_ZIMG)
    target_link_libraries(vmaf_cli PRIVATE PkgConfig::ZIMG)
endif()
